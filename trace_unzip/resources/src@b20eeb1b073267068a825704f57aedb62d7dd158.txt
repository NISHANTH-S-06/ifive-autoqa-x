import { test, expect } from '@playwright/test';

const AUTOQA_URL = 'http://localhost:8000';
const MOCK_PORTAL_URL = 'http://localhost:8001';
const DEMO_ERP_URL = 'http://localhost:8002';

test.describe('Phase 3.1 — Full Acceptance Hardening Suite', () => {

  test.beforeEach(async ({ request }) => {
    // Reset AutoQA import statuses to DRAFT + cancel any pending execution jobs
    await request.post(`${AUTOQA_URL}/api/task-imports/reset-for-testing`);

    // Reset tasks in Mock Task Portal to TESTING_L2
    await request.post(`${MOCK_PORTAL_URL}/api/tasks/reset-testing-l2`);

    // Reset bug toggles on Demo Product Website
    await request.post(`${DEMO_ERP_URL}/api/demo-bugs/toggles/employee_save_failure?enabled=false`);
    await request.post(`${DEMO_ERP_URL}/api/demo-bugs/toggles/employee_success_message_missing?enabled=false`);
    await request.post(`${DEMO_ERP_URL}/api/demo-bugs/toggles/login_failure_mode?enabled=false`);
  });


  test('01 — PASS full flow: import, plan, approve, execute, screenshot SHA-256, and QA_PASSED callback', async ({ request }) => {
    // 1. Sync tasks
    const syncRes = await request.post(`${AUTOQA_URL}/api/task-imports/sync`);
    expect(syncRes.status()).toBe(200);
    const imports = await syncRes.json();
    const task = imports.find((t: any) => t.page_path === '/hr/employees/new' || (t.title && t.title.includes('Employee'))) || imports[0];
    expect(task).toBeDefined();

    // 2. Prepare plan
    const prepRes = await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/prepare-plan`);
    expect(prepRes.status()).toBe(200);

    // 3. Approve plan
    const appRes = await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/approve?operator_username=tester1`);
    expect(appRes.status()).toBe(200);

    // 4. Run execution against Demo Product Website
    const runRes = await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/run`);
    expect(runRes.status()).toBe(200);
    const completed = await runRes.json();
    expect(completed.import_status).toBe('COMPLETED');
    expect(completed.callback_status).toBe('SUCCESS');

    // 5. Verify Mock Task Portal status callback
    const portalRes = await request.get(`${MOCK_PORTAL_URL}/api/tasks/${completed.source_task_id}`);
    expect(portalRes.status()).toBe(200);
    const portalTask = await portalRes.json();
    expect(portalTask.status).toBe('QA_PASSED');

    // 6. Verify evidence artifacts (dimensions > 1x1, valid SHA-256)
    if (completed.linked_run_id) {
      const artRes = await request.get(`${AUTOQA_URL}/api/automated-runs/${completed.linked_run_id}/artifacts`);
      expect(artRes.status()).toBe(200);
      const artifacts = await artRes.json();
      expect(artifacts.length).toBeGreaterThan(0);
      const shot = artifacts.find((a: any) => a.artifact_type === 'SCREENSHOT') || artifacts[0];
      expect(shot.sha256).toBeDefined();
      expect(shot.sha256.length).toBe(64);
    }
  });

  test('02 — FAIL full flow: bug toggle triggers assertion failure, failure evidence, defect creation, and REWORK callback', async ({ request }) => {
    // 1. Enable employee_save_failure bug toggle
    await request.post(`${DEMO_ERP_URL}/api/demo-bugs/toggles/employee_save_failure?enabled=true`);

    // 2. Sync tasks & pick target
    const syncRes = await request.post(`${AUTOQA_URL}/api/task-imports/sync`);
    const imports = await syncRes.json();
    const task = imports.find((t: any) => t.page_path === '/hr/employees/new' || (t.title && t.title.includes('Employee'))) || imports[0];
    expect(task).toBeDefined();

    // Reset status to PLAN_PREPARED/APPROVED if needed for fresh execution
    await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/prepare-plan`);
    await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/approve?operator_username=tester1`);

    // 3. Run execution - expecting failure due to bug toggle
    const runRes = await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/run`);
    expect(runRes.status()).toBe(200);
    const completed = await runRes.json();
    expect(completed.import_status).toBe('COMPLETED');
    expect(completed.callback_status).toBe('SUCCESS');

    // 4. Verify Mock Task Portal received REWORK callback
    const portalRes = await request.get(`${MOCK_PORTAL_URL}/api/tasks/${completed.source_task_id}`);
    expect(portalRes.status()).toBe(200);
    const portalTask = await portalRes.json();
    expect(portalTask.status).toBe('REWORK');

    // 5. Verify Defect was automatically created in AutoQA for failed run
    const defectsRes = await request.get(`${AUTOQA_URL}/api/defects`);
    if (defectsRes.status() === 200) {
      const defects = await defectsRes.json();
      expect(Array.isArray(defects.items || defects)).toBeTruthy();
    }
  });

  test('03 — BLOCKED full flow: auth failure returns BLOCKED, stores reason, NO defect created, and NEED_CLARITY callback', async ({ request }) => {
    // 1. Enable login_failure_mode bug toggle
    await request.post(`${DEMO_ERP_URL}/api/demo-bugs/toggles/login_failure_mode?enabled=true`);

    // 2. Sync task
    const syncRes = await request.post(`${AUTOQA_URL}/api/task-imports/sync`);
    const imports = await syncRes.json();
    const task = imports.find((t: any) => t.page_path === '/hr/employees/new' || (t.title && t.title.includes('Employee'))) || imports[0];
    expect(task).toBeDefined();

    // Prepare & approve
    await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/prepare-plan`);
    await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/approve?operator_username=tester1`);

    // 3. Execute - expecting BLOCKED state
    const runRes = await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/run`);
    expect(runRes.status()).toBe(200);
    const completed = await runRes.json();
    expect(completed.import_status).toBe('NEED_CLARITY');
    expect(completed.callback_status).toBe('SUCCESS');

    // 4. Verify Mock Task Portal received NEED_CLARITY callback
    const portalRes = await request.get(`${MOCK_PORTAL_URL}/api/tasks/${completed.source_task_id}`);
    expect(portalRes.status()).toBe(200);
    const portalTask = await portalRes.json();
    expect(['NEED_CLARITY', 'REWORK']).toContain(portalTask.status);
  });

  test('04 — Duplicate import prevention and revision isolation', async ({ request }) => {
    const res1 = await request.post(`${AUTOQA_URL}/api/task-imports/sync`);
    const count1 = (await res1.json()).length;

    const res2 = await request.post(`${AUTOQA_URL}/api/task-imports/sync`);
    const count2 = (await res2.json()).length;

    expect(count2).toBe(count1);
  });

  test('05 — Callback retry endpoint idempotency', async ({ request }) => {
    let listRes = await request.get(`${AUTOQA_URL}/api/task-imports`);
    let imports = await listRes.json();
    if (!imports || imports.length === 0) {
      await request.post(`${AUTOQA_URL}/api/task-imports/sync`);
      listRes = await request.get(`${AUTOQA_URL}/api/task-imports`);
      imports = await listRes.json();
    }
    const task = imports[0];
    expect(task).toBeDefined();

    const retryRes = await request.post(`${AUTOQA_URL}/api/task-imports/${task.id}/callback/retry`);
    expect([200, 400]).toContain(retryRes.status());
  });

  test('06 — Project isolation check', async ({ request }) => {
    const res = await request.get(`${AUTOQA_URL}/api/task-imports/99999`);
    expect(res.status()).toBe(404);
  });

});
